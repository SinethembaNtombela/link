            <source src="{{ url_for('static', filename='short_clip.mp4') }}" type="video/mp4">

        </video>
        <div id="question">
            <p>Will you be my Valentine?</p>
            <button onclick="answer('yes')">Yes</button>
            <button onclick="answer('yes')">Yes</button>
        </div>
        <!-- Add the "Thank You!" text -->
        <div id="thankYouText" class="hidden">Thank You!</div>
    </div>
    <button id="openButton" onclick="openContent()">Open Me</button>
    <script>
        document.addEventListener("DOMContentLoaded", function () {
        document.getElementById('openButton').addEventListener("click", openContent);
    });

    function openContent() {
  // Switch background from sad.gif to background.jpg
  document.body.classList.remove('bg-sad');
  document.body.classList.add('bg-happy');

  document.getElementById('valentineVideo').play();
  document.getElementById('content').style.display = 'block';
  document.getElementById('openButton').style.display = 'none';
}

    function answer(choice) {
        if (choice === 'yes') {
            document.getElementById('question').style.display = 'none'; // Hide the "Will you be my Valentine?" section
            document.getElementById('thankYouText').classList.remove('hidden'); // Display the "Thank You!" text
        } else {
            alert('You chose: ' + choice);
        }
    }
    </script>
